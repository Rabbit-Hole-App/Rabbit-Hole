-- ONE-OFF DATA REPAIR (not a migration; never in learn-migrations; the .sql.txt name keeps it out of CI's broad **/*.sql class). Owner-run only, by C:/s/spec/r4-quota/repair-roadmap-quota.sh; the lane never runs it.
-- Defect repaired: before the fix, platformCallModel reserved the learner's plan quota BEFORE the prior-call lookup, so a local failure (the over-long LIKE
-- that D1 rejects) burned one unit per outbox retry with nothing dispatched, and past the event's deadline retryRoadmapEvent pushed next_at to now + 3600.
-- Scope: only rows matching both criteria below; every other row is untouched. Idempotent: a second run matches nothing. No row is deleted and no
-- learn_roadmap_calls row is read for anything but counting, so call evidence stays.
--
-- 1. Quota. A reservation is followed by planCall, and planCall precedes every dispatch, so a reservation with NO learn_roadmap_calls row never reached a provider.
--    Per learner plan.hour / plan.day window, units = rows for that learner whose stage is a plan stage (subject, draft, revise), created inside the window and
--    not 'released'. A counter above that count is lowered to it (never raised); a counter at or below it is left alone. Rows whose call is dispatched, unknown or
--    failed after a reply are rows, so their unit stays: unknown outcomes are never refunded.
--    shortcut: a call row created in the first seconds of the window after its reservation was made in the window before shifts one unit between two windows; accepted.
-- 2. Outbox. retryRoadmapEvent sets next_at = now + 3600 only when deadline_at < now, so a backoff inside the deadline never leaves next_at - deadline_at above
--    3600 (it is at most 300). Such a pending plan_requested event of a draft/questioning/proposed roadmap is made due now with a fresh 300 s deadline, unless
--    that roadmap has a call that may have reached a provider (admitted, dispatched, unknown, reconciling): those stay for reconciliation. attempts is kept.
UPDATE learn_roadmap_outbox SET next_at = CAST(strftime('%s', 'now') AS INTEGER), deadline_at = CAST(strftime('%s', 'now') AS INTEGER) + 300
WHERE kind = 'plan_requested' AND processed_at IS NULL
  AND (lease_until IS NULL OR lease_until <= CAST(strftime('%s', 'now') AS INTEGER))
  AND next_at > CAST(strftime('%s', 'now') AS INTEGER) AND next_at - deadline_at > 3600
  AND EXISTS (SELECT 1 FROM learn_roadmaps r WHERE r.id = learn_roadmap_outbox.roadmap_id AND r.state IN ('draft', 'questioning', 'proposed'))
  AND NOT EXISTS (SELECT 1 FROM learn_roadmap_calls c WHERE c.roadmap_id = learn_roadmap_outbox.roadmap_id AND c.checkpoint IN ('admitted', 'dispatched', 'unknown', 'reconciling'));

UPDATE learn_roadmap_quota SET calls = (
  SELECT COUNT(*) FROM learn_roadmap_calls c WHERE c.org = learn_roadmap_quota.org AND c.owner_user_id = learn_roadmap_quota.owner_user_id
    AND c.stage IN ('subject', 'draft', 'revise') AND c.checkpoint <> 'released'
    AND c.created_at >= learn_roadmap_quota.window_start AND c.created_at < learn_roadmap_quota.window_start + CASE learn_roadmap_quota.scope WHEN 'plan.hour' THEN 3600 ELSE 86400 END)
WHERE scope IN ('plan.hour', 'plan.day') AND calls > (
  SELECT COUNT(*) FROM learn_roadmap_calls c WHERE c.org = learn_roadmap_quota.org AND c.owner_user_id = learn_roadmap_quota.owner_user_id
    AND c.stage IN ('subject', 'draft', 'revise') AND c.checkpoint <> 'released'
    AND c.created_at >= learn_roadmap_quota.window_start AND c.created_at < learn_roadmap_quota.window_start + CASE learn_roadmap_quota.scope WHEN 'plan.hour' THEN 3600 ELSE 86400 END);
